cmake_minimum_required(VERSION 3.20)
set(fp_sample "$ENV{ZEPHYR_BASE}/../nrf/samples/bluetooth/fast_pair/locator_tag")
set(KCONFIG_ROOT "${fp_sample}/Kconfig")
set(CONF_FILE "${fp_sample}/configuration/prj.conf;${CMAKE_CURRENT_SOURCE_DIR}/prj.conf")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(rak_findhub)
if(NOT BOARD STREQUAL "rak4631")
  message(FATAL_ERROR "Use rak4631/nrf52840 for the RAK WisMesh Tag")
endif()
set(fp_src "${fp_sample}/src")
target_sources(app PRIVATE main.c locator.c "${fp_src}/ring.c" "${fp_src}/factory_reset.c"
  "${fp_src}/ui/common.c" "../rak_findmy/src/motion.cpp")
target_include_directories(app PRIVATE "${fp_sample}/.." "${fp_sample}/include" "${fp_src}/ui")
zephyr_linker_sources(SECTIONS "${fp_src}/factory_reset_callbacks.ld")
zephyr_linker_sources(DATA_SECTIONS "${fp_src}/ui/request_listener.ld")
include(../rak_findmy/mesh.cmake)
